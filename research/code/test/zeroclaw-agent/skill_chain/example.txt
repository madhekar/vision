To run specific methods with arguments inside a child Python script from a parent script using subprocess, you must expose those internal methods via a Command Line Interface (CLI) in the child script. The subprocess.run() function can only execute files from the operating system level; it cannot target internal Python functions directly without a CLI parser like sys.argv or argparse.Step 1: Set Up the Child Script (script.py)Add a CLI entry point to your child script. import subprocess
import sys

def run_child_method(command_list):
    try:
        # Use sys.executable to ensure the same Python environment is used
        result = subprocess.run(
            [sys.executable, "script.py"] + command_list,
            capture_output=True,
            text=True,
            check=True
        )
        print(f"--- Execution Output ---")
        print(result.stdout)
    except subprocess.CalledProcessError as e:
        print(f"An error occurred: {e.stderr}", file=sys.stderr)

# Example 1: Execute method_one with numerical arguments
print("Calling Method One...")
run_child_method(["method_one", "--x", "10", "--y", "20"])

# Example 2: Execute method_two with string and repeat arguments
print("Calling Method Two...")
run_child_method(["method_two", "--message", "Hello World", "--repeat", "3"])

import sys

def greet(name):
    print(f"Hello, {name}!")

def add(a, b):
    print(int(a) + int(b))

if __name__ == "__main__":
    # sys.argv[1] is the method name, subsequent indexes are arguments
    method_name = sys.argv[1]
    
    if method_name == "greet":
        greet(sys.argv[2])
    elif method_name == "add":
        add(sys.argv[2], sys.argv[3])
