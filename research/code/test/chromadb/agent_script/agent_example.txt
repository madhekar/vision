run_orchistration() {

        echo "⚠️ Warning: Term '$query' contains special characters. Escaping may be needed." >&2

⚠️ Important Security Notes:

Never use shell expansion for untrusted input:

# ❌ UNSAFE (could execute malicious code)
result=$(python3 $SCRIPT "$user_input")

# ✅ SAFE (validate first, quote variables)
if [[ ! "$USER_INPUT" =~ [^a-zA-Z0-9\ -] ]]; then
    python3 $SCRIPT "$USER_INPUT"  # Note: proper quoting! 
fi

🧰 Advanced: Using Python's argparse for robust CLI:

If you want a more polished command-line tool, create /path/to/search_helper.py:

#!/usr/bin/env python3
import subprocess, sys, pathlib, shlex

SC_PATH = "/home/madhekar/.openclaw/workspace/skills/chroma-query-executor/scripts"
SCRIPT_NAME = "skill_orchistration.py"

def search_chroma(term: str):
    if not term or len(term.strip()) == 0:
        print("❌ Error: No search query provided", file=sys.stderr)
        sys.exit(1)
    
    cmd_line = f"python3 {SC_PATH}/{SCRIPT_NAME} {shlex.quote(term)} 2>&1"
    
    # Execute with proper error handling  
    result = subprocess.run(cmd_line, shell=True, capture_output=True, text=True)
    
    if result.returncode != 0:
        print(f"\n⚠️ Warning: Command exited with code {result.returncode}", file=sys.stderr)
        return False
    
    # Print output (if any)  
    if result.stdout.strip():
        print(result.stdout, end="")
    
    if result.stderr.strip():  # Some errors are expected and okay
        print(f"\nStderr: {result.stderr}\n", file=sys.stderr)
    
    return True

if __name__ == "__main__":
    term = sys.argv[1] if len(sys.argv) > 1 else ""
    search_chroma(term or "default_search_term")
